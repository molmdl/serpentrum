```python
def broaden(modes, fwhm=16.0, x_min=0.0, x_max=None, n_points=800):
    """Sum of Gaussians, one per mode: y(x) = sum(I_i * exp(-0.5*((x-f_i)/sigma)^2)).
    sigma = fwhm / (2*sqrt(2*ln(2)))  (~ fwhm/2.35482).
    Grid: xs[i] = x_min + (x_max - x_min) * i / (n_points - 1), i in 0..n_points-1
    (both ends inclusive). x_max default: max(3600.0, max(freq) + 5*sigma) — the
    3600 floor keeps the plotted axis at the xtb display range; negatives are NOT
    special-cased (their in-grid tail is negligible: exp(-11) at -31.9 with
    fwhm 16); zero-intensity modes contribute exactly 0 by arithmetic.
    Raises SpectraParseError('broaden: empty mode list') on no modes (defensive),
    ValueError on fwhm <= 0 or n_points < 2."""
```
