```python
def broaden(modes, fwhm=16.0, x_min=0.0, x_max=None, n_points=800):
    """Sum of Gaussians, one per mode: y(x) = sum(I_i * exp(-0.5*((x-f_i)/sigma)^2)).
    sigma = fwhm / (2*sqrt(2*ln(2)))  (~ fwhm/2.35482; 16 -> 6.794574, 32 -> 13.58915).
    Grid: xs[i] = x_min + (x_max - x_min) * i / (n_points - 1), i in 0..n_points-1
    (both ends inclusive). x_max default: max(3600.0, max(freq) + 5*sigma) — the
    3600 floor keeps the plotted axis at the xtb display range; with NO modes the
    default grid is [x_min, 3600.0].
    EMPTY MODES -> (grid, [0.0]*n_points): the PINNED zero-curve semantics
    (a silent spectrum plots as a flat zero line, never an exception).
    Negatives are NOT special-cased (their in-grid tail is negligible: exp(-11)
    at -31.9 with fwhm 16); zero-intensity modes contribute exactly 0 by arithmetic.
    Raises ValueError on fwhm <= 0 or n_points < 2."""
```
