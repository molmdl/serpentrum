## Method

**Algorithm replication (no import of `serpentrum/wall.py`; verified against wall.py:230-279).**
Per attempt, the PINNED draw order: anchor x uniform over the margin-shrunk box, anchor y
likewise, theta uniform over `[0, pi)`; draws consumed on EVERY attempt regardless of leg
outcome. Legs in order: **L1** both endpoints inside the margin-shrunk box; **L2** every
new-mol atom >= 3.0 (per-atom); **L3** head >= 8.0 (squared compare); **L4** every other
live mol's atom >= 3.0 (skipped when empty); **NEW L5** candidate keeps >= 8.0 TRUE
segment-segment distance from every existing wall (a proper crossing => distance 0 =>
reject; `_segments_intersect` + the min of four point-segment distances). Constants block
mirrored from wall.py:64-74; the three 9.1 drafts (`WALL_MAX_LIVE_WALLS=5`,
`WALL_ACCUM_LENGTH_FRACTION=0.35`, `WALL_WALL_CLEARANCE_A=8.0`). **Two-tier length:**
`half_len = 0.5 * fraction * min(x-span, y-span)` with fraction = 0.45 when
`len(existing)==0` else 0.35 (computed once per call). **Cap gate:** `len(existing) >= 5`
=> `None` consuming ZERO draws (degenerate-guard pattern, wall.py:236-239).
**WALL_MAX_DRAWS = 32** attempts before None.

**Fixtures.** Adversarial standing sets (>= 2 distinct arrangements per k): A-parallel
(evenly spread horizontal stripes — maximal coverage of the drawable lattice),
B-pinwheel (k segments through a common centre at distinct angles), C-faces (staggered
near-opposite-face segments); plus `empty` at k=0. Head/mol fixtures: **far** (head at a
corner, new mol at the opposite corner, no live mols => isolates L1) and **close** (head
at centre, new mol ~12 A away, two live mols). **Realism arm:** a seeded accumulate
simulation (seed 4242+k) with a drifting head, a new mol near the head, and growing live
mol atoms, then the NEXT call's acceptance measured at that accumulated state.

**Draw budget.** >= 12000 candidate draws per main state (plan minimum 10000); 6000 per
mitigation-grid state. Verdict rubric keyed ONLY on the in-budget metric
`P(exhaustion in 32) = (1-p)^32`; per-draw p is diagnostic. Exhaustion => `None` is
BENIGN by design (the caller keeps the accumulated set unchanged).

```python
"""Phase 9.1 research probe: accumulating line-wall feasibility (Monte-Carlo).

Self-contained PURE python3.6 probe (stdlib math/random only). REPLICATES the
draw+leg algorithm of serpentrum/wall.py:230-279 plus the 9.1 DRAFT accumulation
surface (cap gate, two-tier length, wall-vs-wall leg L5) -- it does NOT import
wall.py (at Wave-1 start the shipped module has no accumulation surface yet).

Measures, per box preset x accumulated count k, the per-draw acceptance p of a
new wall and the IN-BUDGET exhaustion metric P(exhaustion in WALL_MAX_DRAWS=32)
= (1-p)^32 -- against adversarial standing-wall fixtures AND a seeded realistic
accumulate simulation. Verdicts on the three 9.1 draft constants are keyed ONLY
on the in-budget metric; per-draw p is diagnostic context.

No repo file is modified by this probe.
"""
import math
import random

# --- constants mirrored from serpentrum/wall.py:64-74 --------------------
WALL_LENGTH_FRACTION = 0.45
WALL_BOX_MARGIN_A = 6.0
WALL_HEAD_CLEARANCE_A = 8.0
WALL_MOL_CLEARANCE_A = 3.0
WALL_LIVE_CLEARANCE_A = 3.0
WALL_MAX_DRAWS = 32

# --- Phase 9.1 DRAFT constants (the ones under test) ---------------------
WALL_MAX_LIVE_WALLS = 5
WALL_ACCUM_LENGTH_FRACTION = 0.35
WALL_WALL_CLEARANCE_A = 8.0

PRESETS = {
    'small': ((-35.0, -35.0), (35.0, 35.0)),
    'medium': ((-55.0, -55.0), (55.0, 55.0)),
    'large': ((-85.0, -85.0), (85.0, 85.0)),
}
NDRAW = 12000   # >= 10,000 per state (plan requirement) for the main tables
NGRID = 6000    # supplementary mitigation-grid draws


# --- geometry helpers (mirror wall.py:90-141 + a true segment-segment L5) -
def _seg_point_dist_sq(px, py, ax, ay, bx, by):
    dx = bx - ax
    dy = by - ay
    seg_sq = dx * dx + dy * dy
    if seg_sq <= 0.0:
        ex = px - ax
        ey = py - ay
        return ex * ex + ey * ey
    t = ((px - ax) * dx + (py - ay) * dy) / seg_sq
    if t < 0.0:
        t = 0.0
    elif t > 1.0:
        t = 1.0
    cx = ax + t * dx
    cy = ay + t * dy
    ex = px - cx
    ey = py - cy
    return ex * ex + ey * ey


def _orient(ax, ay, bx, by, cx, cy):
    return (bx - ax) * (cy - ay) - (by - ay) * (cx - ax)


def _on_seg(ax, ay, bx, by, cx, cy):
    return (min(ax, bx) - 1e-12 <= cx <= max(ax, bx) + 1e-12 and
            min(ay, by) - 1e-12 <= cy <= max(ay, by) + 1e-12)


def _segments_intersect(p1, p2, p3, p4):
    ax, ay = p1
    bx, by = p2
    cx, cy = p3
    dx, dy = p4
    d1 = _orient(ax, ay, bx, by, cx, cy)
    d2 = _orient(ax, ay, bx, by, dx, dy)
    d3 = _orient(cx, cy, dx, dy, ax, ay)
    d4 = _orient(cx, cy, dx, dy, bx, by)
    if ((d1 > 0 and d2 < 0) or (d1 < 0 and d2 > 0)) and \
       ((d3 > 0 and d4 < 0) or (d3 < 0 and d4 > 0)):
        return True
    if d1 == 0 and _on_seg(ax, ay, bx, by, cx, cy):
        return True
    if d2 == 0 and _on_seg(ax, ay, bx, by, dx, dy):
        return True
    if d3 == 0 and _on_seg(cx, cy, dx, dy, ax, ay):
        return True
    if d4 == 0 and _on_seg(cx, cy, dx, dy, bx, by):
        return True
    return False


def seg_seg_dist_sq(p1, p2, p3, p4):
    """True squared segment-to-segment distance (proper crossing => 0)."""
    if _segments_intersect(p1, p2, p3, p4):
        return 0.0
    ax, ay = p1
    bx, by = p2
    cx, cy = p3
    dx, dy = p4
    return min(_seg_point_dist_sq(ax, ay, cx, cy, dx, dy),
               _seg_point_dist_sq(bx, by, cx, cy, dx, dy),
               _seg_point_dist_sq(cx, cy, ax, ay, bx, by),
               _seg_point_dist_sq(dx, dy, ax, ay, bx, by))


def min_atom_seg_dist(endpoints, atoms):
    (ax, ay), (bx, by) = endpoints
    best = None
    for a in atoms:
        if len(a) == 4:
            px, py = a[1], a[2]
        else:
            px, py = a[0], a[1]
        d = math.sqrt(_seg_point_dist_sq(px, py, ax, ay, bx, by))
        if best is None or d < best:
            best = d
    return best


def mol_atoms(cx, cy, radius=1.4, n=6):
    return [(radius * math.cos(k * 2.0 * math.pi / n) + cx,
             radius * math.sin(k * 2.0 * math.pi / n) + cy)
            for k in range(n)]


# --- algorithm replication (wall.py:230-279 + 9.1 draft) -----------------
def _shrunk(box):
    (x0, y0), (x1, y1) = box
    m = WALL_BOX_MARGIN_A
    return x0 + m, x1 - m, y0 + m, y1 - m


def _half_len(box, frac, n_existing):
    (x0, y0), (x1, y1) = box
    f = WALL_LENGTH_FRACTION if n_existing == 0 else frac
    return 0.5 * f * min(x1 - x0, y1 - y0)


def _draw(rng, sx0, sx1, sy0, sy1, half_len):
    cx = rng.uniform(sx0, sx1)
    cy = rng.uniform(sy0, sy1)
    theta = rng.uniform(0.0, math.pi)
    ct = math.cos(theta)
    st = math.sin(theta)
    return ((cx - half_len * ct, cy - half_len * st),
            (cx + half_len * ct, cy + half_len * st))


def _check(ep, existing, head, new_atoms, live_atoms, sx0, sx1, sy0, sy1,
           wwall):
    (ax, ay), (bx, by) = ep
    # L1 both endpoints inside the margin-shrunk box on both axes
    if ax < sx0 or ax > sx1 or ay < sy0 or ay > sy1:
        return 'L1'
    if bx < sx0 or bx > sx1 or by < sy0 or by > sy1:
        return 'L1'
    # L2 every new-mol atom >= 3.0 (PER-ATOM)
    if min_atom_seg_dist(ep, new_atoms) < WALL_MOL_CLEARANCE_A:
        return 'L2'
    # L3 head >= 8.0 (squared compare)
    if _seg_point_dist_sq(head[0], head[1], ax, ay, bx, by) \
            < WALL_HEAD_CLEARANCE_A * WALL_HEAD_CLEARANCE_A:
        return 'L3'
    # L4 every other live mol's atom >= 3.0 (skip when empty)
    if live_atoms and min_atom_seg_dist(ep, live_atoms) < WALL_LIVE_CLEARANCE_A:
        return 'L4'
    # L5 candidate must keep >= wwall from every existing wall
    for (w1, w2) in existing:
        if seg_seg_dist_sq((ax, ay), (bx, by), w1, w2) < wwall * wwall:
            return 'L5'
    return None


def sample_one(existing, head, new_atoms, live_atoms, box, rng, cap,
               frac_acc, max_draws, wwall=WALL_WALL_CLEARANCE_A):
    """Algorithm-exact single next_wall call (returns endpoints or None)."""
    (x0, y0), (x1, y1) = box
    m = WALL_BOX_MARGIN_A
    if (x1 - x0) < 2.0 * m or (y1 - y0) < 2.0 * m:
        return None
    if len(existing) >= cap:            # cap gate: ZERO draws
        return None
    sx0, sx1, sy0, sy1 = _shrunk(box)
    half_len = _half_len(box, frac_acc, len(existing))
    for _ in range(max_draws):
        ep = _draw(rng, sx0, sx1, sy0, sy1, half_len)
        if _check(ep, existing, head, new_atoms, live_atoms,
                  sx0, sx1, sy0, sy1, wwall) is None:
            return ep
    return None


def measure(existing, head, new_atoms, live_atoms, box, frac_acc, rng,
            wwall=WALL_WALL_CLEARANCE_A, ndraw=NDRAW):
    """Monte-Carlo per-draw acceptance + first-failing-leg breakdown."""
    sx0, sx1, sy0, sy1 = _shrunk(box)
    half_len = _half_len(box, frac_acc, len(existing))
    counts = {'accept': 0, 'L1': 0, 'L2': 0, 'L3': 0, 'L4': 0, 'L5': 0}
    for _ in range(ndraw):
        ep = _draw(rng, sx0, sx1, sy0, sy1, half_len)
        r = _check(ep, existing, head, new_atoms, live_atoms,
                   sx0, sx1, sy0, sy1, wwall)
        if r is None:
            counts['accept'] += 1
        else:
            counts[r] += 1
    p = counts['accept'] / float(ndraw)
    return p, counts


def exhaustion(p):
    return (1.0 - p) ** WALL_MAX_DRAWS


# --- fixtures ------------------------------------------------------------
def adversarial_sets(box, k, frac_acc):
    """>=2 hand-designed adversarial standing-wall arrangements for count k."""
    if k == 0:
        return [('empty', [])]
    (x0, y0), (x1, y1) = box
    sx0, sx1, sy0, sy1 = _shrunk(box)
    span = min(x1 - x0, y1 - y0)
    L0 = WALL_LENGTH_FRACTION * span
    L1 = frac_acc * span

    def length_for(i):
        return L0 if i == 0 else L1

    # A: parallel horizontal stripes evenly spread in y -> maximal coverage of
    #    the drawable lattice ('shrink the remaining lattice' adversarial).
    A = []
    for i in range(k):
        L = length_for(i)
        y = sy0 + (i + 0.5) * (sy1 - sy0) / k
        A.append(((0.0 - L / 2.0, y), (0.0 + L / 2.0, y)))

    # B: pinwheel -- k segments through a common center at distinct angles.
    B = []
    cxm = 0.5 * (sx0 + sx1)
    cym = 0.5 * (sy0 + sy1)
    for i in range(k):
        L = length_for(i)
        theta = (i * math.pi) / k
        ct = math.cos(theta)
        st = math.sin(theta)
        B.append(((cxm - L / 2.0 * ct, cym - L / 2.0 * st),
                  (cxm + L / 2.0 * ct, cym + L / 2.0 * st)))

    # C: staggered near-opposite-face segments, alternating top/bottom faces.
    C = []
    for i in range(k):
        L = length_for(i)
        y = sy1 - 1.0 if i % 2 == 0 else sy0 + 1.0
        off = (i // 2) * (L * 0.5)
        xc = cxm + (off if i % 4 < 2 else -off)
        C.append(((xc - L / 2.0, y), (xc + L / 2.0, y)))
    return [('A-parallel', A), ('B-pinwheel', B), ('C-faces', C)]


def far_fixture(box):
    (x0, y0), (x1, y1) = box
    head = (x1 - 4.0, y1 - 4.0)
    atoms = mol_atoms(x0 + 5.0, y0 + 5.0)  # new mol far, opposite corner
    return head, atoms, []


def close_fixture(box):
    (x0, y0), (x1, y1) = box
    head = (0.0, 0.0)
    atoms = mol_atoms(12.0, 0.0)           # new mol ~12 A from head
    live = mol_atoms(x0 + 10.0, y0 + 10.0) + mol_atoms(x1 - 12.0, y1 - 15.0)
    return head, atoms, live


def realistic_sequence(box, count):
    """A capture sequence: head drifts, new mol near head, live mols grow."""
    (x0, y0), (x1, y1) = box
    span = x1 - x0
    fixtures = []
    live = []
    for i in range(count):
        t = (i + 0.5) / float(count)
        hx = x0 + 0.20 * span + t * 0.60 * span
        hy = y0 + 0.25 * span + 0.5 * span * math.sin(3.0 * t)
        head = (hx, hy)
        ang = 1.1 * (i + 1)
        mx = hx + 13.0 * math.cos(ang)
        my = hy + 13.0 * math.sin(ang)
        mx = max(x0 + 4.0, min(x1 - 4.0, mx))
        my = max(y0 + 4.0, min(y1 - 4.0, my))
        atoms = mol_atoms(mx, my)
        fixtures.append((head, atoms, list(live)))
        live = live + atoms
    return fixtures


def realism_state(box, k, seed, cap, frac_acc, wwall=WALL_WALL_CLEARANCE_A):
    rng = random.Random(seed)
    existing = []
    fixtures = realistic_sequence(box, k + 1)
    for i in range(k):
        head, atoms, live = fixtures[i]
        w = sample_one(existing, head, atoms, live, box, rng, cap,
                       frac_acc, WALL_MAX_DRAWS, wwall)
        if w is not None:
            existing.append(w)
    head, atoms, live = fixtures[k]
    return existing, head, atoms, live


# --- evaluation ----------------------------------------------------------
def eval_adv(box, k, frac_acc, cap, wwall, ndraw=NDRAW):
    """All adversarial (arr, fixture) states -> min-p per fixture class."""
    rng = random.Random(90210 + k + int(frac_acc * 100.0))
    far = None
    close = None
    worst = None
    for label, walls in adversarial_sets(box, k, frac_acc):
        for fxname, fx in (('far', far_fixture(box)),
                           ('close', close_fixture(box))):
            head, atoms, live = fx
            p, counts = measure(walls, head, atoms, live, box, frac_acc,
                                rng, wwall, ndraw)
            rec = (p, counts, label, fxname)
            if worst is None or p < worst[0]:
                worst = rec
            if fxname == 'far' and (far is None or p < far[0]):
                far = rec
            if fxname == 'close' and (close is None or p < close[0]):
                close = rec
    return worst, far, close


def verdict_label(exh, vanished):
    if vanished or exh > 0.10:
        return 'RED'
    if exh > 0.01:
        return 'AMBER'
    return 'GREEN'


def cap_worst(box, cap, frac_acc=WALL_ACCUM_LENGTH_FRACTION,
              wwall=WALL_WALL_CLEARANCE_A, ndraw=NDRAW):
    """Worst adversarial P_exh over k=0..cap-1 for a cap/fraction/wwall bundle."""
    worst_exh = 0.0
    worst_p = 1.0
    van = False
    for k in range(cap):
        w, far, close = eval_adv(box, k, frac_acc, cap, wwall, ndraw)
        p, counts = w[0], w[1]
        if p < worst_p:
            worst_p = p
        worst_exh = max(worst_exh, exhaustion(p))
        if counts['accept'] == 0:
            van = True
    return worst_exh, worst_p, van
```
